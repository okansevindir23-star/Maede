        <div class="info">📍 Boylam (Longitude): <span id="boylam" class="highlight">-</span></div>

    <script>
        window.onload = function() {
            if (navigator.geolocation) {
                // Tarayıcıdan hassas konum (GPS) istiyoruz
                navigator.geolocation.getCurrentPosition(konumuBasariylaAl, hataVer, {
                    enableHighAccuracy: true, // En yüksek GPS hassasiyeti
                    timeout: 10000,
                    maximumAge: 0
                });
            } else {
                document.getElementById("durum").innerText = "Tarayıcınız GPS özelliğini desteklemiyor.";
            }
        };

        function konumuBasariylaAl(position) {
            const lat = position.coords.latitude;
            const lon = position.coords.longitude;
            const accuracy = position.coords.accuracy;

            // Ekrana yazdır
            document.getElementById("durum").innerText = "Konum başarıyla alındı!";
            document.getElementById("enlem").innerText = lat;
            document.getElementById("boylam").innerText = lon;
            document.getElementById("hassasiyet").innerText = Math.round(accuracy);

            // Google Maps bağlantısını oluştur ve butonu görünür yap
            const mapsUrl = `https://www.google.com/maps?q=${lat},${lon}`;
            const btn = document.getElementById("haritaLinki");
            btn.href = mapsUrl;
            btn.style.display = "block";
            btn.innerText = "Haritada Aç (Nokta Atışı Gör)";
        }

        function hataVer(error) {
            if(error.code === error.PERMISSION_DENIED) {
                document.getElementById("durum").innerText = "Konum izni reddedildi! Projede test etmek için izin vermelisiniz.";
                document.getElementById("durum").style.color = "red";
            } else {
                document.getElementById("durum").innerText = "Konum alınırken bir hata oluştu.";
            }
        }
    </script>
